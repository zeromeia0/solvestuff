\ficha{Unidades de Medida e Conversões}

\instrucoes{Resolve os exercícios, apresentando os cálculos e indicando sempre as unidades. Lembra-te: $1\,\text{dm}^3 = 1\,\text{L}$.}

\secao{Comprimento}{Converte para a unidade indicada.}
\begin{exercicios}
  \item 3,5\,m em cm
  \item 420\,cm em m
  \item 2,8\,km em m
  \item 750\,m em km
  \item 65\,mm em cm
  \item 0,4\,dm em mm
  \item 12\,dam em dm
  \item 3\,km 250\,m em km
  \item Ordena por ordem crescente: 1,2\,km; 1\,150\,m; 118\,dam; 12\,000\,dm.
  \item Uma pista tem 400\,m. Quantas voltas são precisas para fazer 10\,km?
\end{exercicios}

\secao{Massa e Capacidade}{Converte para a unidade indicada.}
\begin{exercicios}
  \item 2,5\,kg em g
  \item 750\,g em kg
  \item 3\,t em kg
  \item 45\,mg em g
  \item 1,5\,L em mL
  \item 250\,mL em L
  \item 3,2\,dL em cL
  \item 0,75\,hL em L
  \item Quantos copos de 200\,mL se enchem com uma garrafa de 1,5\,L?
  \item Um pacote de arroz tem 1\,kg. Quantas porções de 75\,g dá, aproximadamente?
\end{exercicios}

\secao{Área}{Converte para a unidade indicada. Lembra-te: $1\,\text{m}^2 = 100\,\text{dm}^2$.}
\begin{exercicios}
  \item 3\,m$^2$ em dm$^2$
  \item 5\,m$^2$ em cm$^2$
  \item 450\,cm$^2$ em dm$^2$
  \item 28\,000\,cm$^2$ em m$^2$
  \item 2,5\,km$^2$ em m$^2$
  \item 1\,ha em m$^2$ (1 hectare = 1\,hm$^2$)
  \item 35\,000\,m$^2$ em hectares
  \item 0,8\,dm$^2$ em mm$^2$
  \item Um terreno retangular mede 250\,m por 120\,m. Qual é a área em hectares?
  \item Uma folha A4 mede 21\,cm por 29,7\,cm. Qual é a área em dm$^2$?
\end{exercicios}

\secao{Volume e Capacidade}{Converte para a unidade indicada. Lembra-te: $1\,\text{m}^3 = 1\,000\,\text{dm}^3$.}
\begin{exercicios}
  \item 2\,m$^3$ em dm$^3$
  \item 4\,500\,cm$^3$ em dm$^3$
  \item 0,3\,m$^3$ em litros
  \item 750\,cm$^3$ em mL
  \item 12\,L em cm$^3$
  \item 25\,000\,L em m$^3$
  \item 1,2\,dm$^3$ em cm$^3$
  \item Um aquário tem 60\,cm $\times$ 30\,cm $\times$ 40\,cm. Quantos litros de água leva?
  \item Uma piscina tem 10\,m $\times$ 5\,m $\times$ 1,6\,m. Quantos litros de água leva quando cheia?
  \item Quantos cubos de 1\,cm$^3$ cabem num cubo de 1\,dm$^3$?
\end{exercicios}

\secao{Tempo}{Converte ou calcula.}
\begin{exercicios}
  \item 3 horas em minutos.
  \item 150 minutos em horas e minutos.
  \item 2,5 horas em horas e minutos.
  \item 1 hora e 45 minutos em horas (na forma decimal).
  \item 5\,400 segundos em horas.
  \item Um dia em segundos.
  \item Um filme começou às 21h35 e durou 2h18min. A que horas acabou?
  \item Um comboio partiu às 8h47 e chegou às 11h12. Quanto tempo demorou a viagem?
\end{exercicios}

\secao{Desafios}{Resolve os problemas.}
\begin{exercicios}
  \item Uma torneira pinga 1 gota por segundo e cada gota tem 0,05\,mL. Quantos litros se perdem num dia?
  \item Um terreno de 2,4\,ha foi dividido em 8 lotes iguais. Qual é a área de cada lote, em m$^2$?
  \item Um depósito cúbico tem 1,5\,m de aresta. Quantos garrafões de 5\,L são precisos para o encher?
  \item Explica por que razão, ao converter de m$^2$ para cm$^2$, se multiplica por 10\,000 e não por 100.
  \item Um atleta correu uma maratona (42,195\,km) em 2h05min. Qual foi a velocidade média em km/h? (Arredonda às décimas.)
  \item Uma caixa de leite tem 1\,L e forma de paralelepípedo com base de 6\,cm por 9\,cm. Qual é a altura mínima da caixa? (Arredonda às décimas.)
\end{exercicios}
