\ficha{Monómios e Polinómios}[Expressões Algébricas]

\instrucoes{Simplifica sempre as expressões e escreve os polinómios na forma reduzida, ordenados por potências decrescentes.}

\secao{Expressões Algébricas}{Traduz em linguagem algébrica ou calcula o valor numérico.}
\begin{exercicios}
  \item O dobro de um número $x$.
  \item Um número $x$ mais 5.
  \item O triplo de um número $y$ menos 4.
  \item O quadrado de um número $a$ mais o seu dobro.
  \item A soma de dois números consecutivos, sendo $n$ o menor.
  \item O perímetro de um retângulo de comprimento $c$ e largura $\ell$.
  \item Calcula $3x + 2$ para $x = 4$ e para $x = -1$.
  \item Calcula $a^2 - 2a$ para $a = 3$ e para $a = -2$.
  \item Calcula $2xy - y^2$ para $x = 1$ e $y = -3$.
  \item Calcula $\dfrac{x + y}{2}$ para $x = 5$ e $y = -9$.
\end{exercicios}

\secao{Monómios}{Indica o coeficiente, a parte literal e o grau de cada monómio.}
\begin{exercicios}
  \item $5x^2$
  \item $-3ab$
  \item $\dfrac{2}{3}x^3y$
  \item $-x^4$
  \item $7$
  \item $0{,}5\,a^2b^3$
  \item Indica dois monómios semelhantes a $4x^2y$.
  \item Os monómios $3x^2y$ e $3xy^2$ são semelhantes? Justifica.
\end{exercicios}

\secao{Operações com Monómios}{Calcula e simplifica.}
\begin{exercicios}
  \item $3x + 5x$
  \item $7a^2 - 2a^2$
  \item $4xy - 9xy + xy$
  \item $2x \times 3x$
  \item $(-4a^2) \times 5a^3$
  \item $\dfrac{1}{2}x^2y \times 6xy$
  \item $12x^5 : 3x^2$
  \item $(-15a^4b^2) : (5a^2b)$
  \item $(3x^2)^2$
  \item $(-2ab^3)^3$
\end{exercicios}

\secao{Adição e Subtração de Polinómios}{Simplifica.}
\begin{exercicios}
  \item $(3x + 2) + (5x - 7)$
  \item $(4x^2 - 3x + 1) + (x^2 + 5x - 4)$
  \item $(6x - 5) - (2x + 3)$
  \item $(2x^2 + x - 1) - (x^2 - 3x + 4)$
  \item $5a - (3a - 2) + (a + 1)$
  \item $(x^3 - 2x) - (3x^3 + x^2 - 2x)$
  \item $-(2y - 3) - (5 - y)$
  \item $\dfrac{1}{2}x - \left(\dfrac{1}{3}x - 1\right)$
\end{exercicios}

\secao{Multiplicação de Polinómios}{Aplica a propriedade distributiva e simplifica.}
\begin{exercicios}
  \item $3(x + 4)$
  \item $-2(3x - 5)$
  \item $x(x - 6)$
  \item $2a(a^2 - 3a + 1)$
  \item $(x + 2)(x + 5)$
  \item $(x - 3)(x + 4)$
  \item $(2x - 1)(x - 3)$
  \item $(3a + 2)(2a - 5)$
  \item $(x + 1)(x^2 - x + 1)$
  \item $2(x - 1) - 3(x + 2)$
  \item $x(x + 3) - (x + 1)(x - 2)$
\end{exercicios}

\secao{Desafios}{Resolve os problemas.}
\begin{exercicios}
  \item Um retângulo tem comprimento $2x + 3$ e largura $x - 1$. Escreve polinómios para o perímetro e para a área.
  \item Determina o polinómio $P$ tal que $P + (3x^2 - x + 2) = 5x^2 + 4x - 1$.
  \item O lado de um quadrado é $x + 4$. Escreve a área e simplifica. Qual é a área quando $x = 2$?
  \item Mostra que $(n + 1)^2 - n^2$ é sempre um número ímpar, para $n$ natural.
  \item Simplifica $(x + 2)(x - 2) - (x - 1)^2$.
  \item Um terreno quadrado de lado $x$ perdeu uma faixa de 3\,m de largura num dos lados. Escreve a área restante como polinómio.
\end{exercicios}
