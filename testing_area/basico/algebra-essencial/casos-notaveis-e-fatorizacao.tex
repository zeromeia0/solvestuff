\ficha{Casos Notáveis e Fatorização}

\instrucoes{Casos notáveis: $(a + b)^2 = a^2 + 2ab + b^2$, $\;(a - b)^2 = a^2 - 2ab + b^2$ e $(a + b)(a - b) = a^2 - b^2$. Simplifica sempre os resultados.}

\secao{Quadrado da Soma e da Diferença}{Desenvolve, aplicando os casos notáveis.}
\begin{exercicios}
  \item $(x + 3)^2$
  \item $(x - 5)^2$
  \item $(a + 1)^2$
  \item $(2x + 1)^2$
  \item $(3x - 2)^2$
  \item $(x + \tfrac{1}{2})^2$
  \item $(5 - y)^2$
  \item $(2a + 3b)^2$
  \item $(x^2 - 4)^2$
  \item $(-x - 7)^2$
\end{exercicios}

\secao{Diferença de Quadrados}{Desenvolve, aplicando os casos notáveis.}
\begin{exercicios}
  \item $(x + 4)(x - 4)$
  \item $(a - 9)(a + 9)$
  \item $(2x + 3)(2x - 3)$
  \item $(5 - x)(5 + x)$
  \item $(3a - 1)(3a + 1)$
  \item $(x^2 + 1)(x^2 - 1)$
  \item $\left(\tfrac{1}{2}x - 2\right)\left(\tfrac{1}{2}x + 2\right)$
  \item Calcula mentalmente $21 \times 19$ usando $(20 + 1)(20 - 1)$.
  \item Calcula mentalmente $102 \times 98$.
\end{exercicios}

\secao{Expressões e Equações}{Simplifica ou resolve, usando os casos notáveis e a fatorização.}
\begin{exercicios}
  \item $(x + 2)^2 - (x - 2)^2$
  \item $(x + 1)^2 - (x + 3)(x - 3)$
  \item $(2x - 1)^2 - 4x(x - 1)$
  \item Resolve: $x^2 - 16 = 0$
  \item Resolve: $x^2 + 8x + 16 = 0$
  \item Resolve: $x^2 - 5x = 0$
  \item Resolve: $4x^2 - 9 = 0$
  \item Resolve: $(x - 3)^2 = 25$
\end{exercicios}

\secao{Desafios}{Resolve os problemas.}
\begin{exercicios}
  \item Calcula $999^2$ sem calculadora, usando $(1000 - 1)^2$.
  \item Calcula $57^2 - 43^2$ sem calculadora.
  \item Um quadrado de lado $x + 5$ tem a mais quanto de área do que um quadrado de lado $x$? Simplifica.
  \item Mostra que $(n + 1)^2 - (n - 1)^2 = 4n$.
  \item Determina $k$ de modo que $x^2 + kx + 49$ seja o quadrado de um binómio.
  \item Mostra que o produto de dois números ímpares consecutivos mais 1 é sempre um quadrado perfeito.
\end{exercicios}
